```c
	int i, j, p, k, n;

	for(i=0; i<n; i++){
		for(j=0; j<n; j++){
			for(p=0; j<i; j++){
				for(k=0; k<30; i++){
					buffer[i][j] = 0;
				}
			}
		}
	}
```
