```c
char check_cycle(int adj_matrix[6][6], char *path) {
    int len, i, snode, enode;
    int a_m[6][6] = { 0 };  //간선을 지나간 적이 있는지 체크하기 위해서 사용

    len = strlen(path);

    // 시작정점과 종료정점이 다르면 ‘n'을 리턴
    if (path[0] != path[len - 1])                return 'n';

    for (i = 0; i < len - 1 ; i++){
        snode = path[i] - '0';
        enode = path[i + 1] - '0';

        // 두 정점이 연결되지 않으면 (간선이 없으면) ‘n'을 리턴
        if (adj_matrix[snode][enode] != 1)   return 'n';

        // 두 정점이 간선으로 연결되어 있지만 이 간선을 전에
        // 지나간 적이 있으면 ‘n'을 리턴
        if (a_m[snode][enode] == 1)         return 'n';
        else if (a_m[enode][snode] == 1)    return 'n';
        else {
            // 이 간선을 지나간 적이 없으면 이 번에 지나가면서
            // 지나간 것을 표시 (양 방향)
            a_m[snode][enode] = 1;
            a_m[enode][snode] = 1;
        }
    }

    return 'y';
}
```
(주어진 main 실행 결과: Cycle?: Yes)

## 9번

9. 다음의 코드에서 배열 list는 50개의 난수(2에서 50사이의 난수)들을 저장하고 있으며 배열 list에서 소수(prime number)들의 합을 구하는 문제. 구체적인 조건은 가장 작은 소수들부터 차례로 더하여 소수들의 합을 구하였을 경우에 그 합이 50(<50)을 넘지 않는 가장 큰 합을 구하시오. [8 points]

```c
#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#define MAX_SIZE 50
int main(){
    int i;
    int list[MAX_SIZE];
    /* 빈칸 */

    srand((unsigned)time(NULL));
    for (i = 0; i < MAX_SIZE i++)  list[i] = rand() % 49 + 2;
    /* 빈칸 */
}
```

**원본 답:**

```c
    int j, smaller, temp, sum = 0, prime_flag;
    /* (srand, 난수 생성 for 는 주어진 부분) */
    // 난수들을 오름차순으로 정렬
    for (i = 0; i < MAX_SIZE - 1; i++) {
        smaller = i;
        for (j = i + 1; j<MAX_SIZE j++)
              if (list[j] < list[smaller]) smaller = j;

        temp = list[i];
        list[i] = list[smaller];
        list[smaller] = temp;
    }

    for (i = 0; i < MAX_SIZE i++){
        //정렬된 수를 가장 작은 수부터 소수인지 확인
        prime_flag = 1;
        for (j = 2; j < list[i]; j++)
             if (list[i] % j == 0) prime_flag = 0;

        // 소수이면 합이 50을 넘지 않을 때까지 더한다.
        if (prime_flag == 1)
            if (sum + list[i] < 50) sum = sum + list[i];
    }
    printf("sum:%d \n", sum);
    return 0;
}
```
