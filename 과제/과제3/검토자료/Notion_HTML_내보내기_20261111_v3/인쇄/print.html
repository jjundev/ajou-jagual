구현 목적

퀀텀 2의 라운드로빈(Round-Robin) 준비 큐와 중단 문맥 스택으로, 하나의 CPU에서 프로세스와 중첩 인터럽트가 실행되는 순서를 tick 단위로 시뮬레이션한다.

소스코드 분석

HW3_skeleton.c의 선언·제공 함수·main은 그대로 두고, 비어 있던 select_process, start_interrupt, finish_tick을 구현했다.

헤더·매크로·구조체·함수 원형

소스코드기능 명세와 분석
#define _CRT_SECURE_NO_WARNINGS

#include <stdio.h>
#include <stdlib.h>

#define MAX_PROCESSES 8
#define MAX_INTERRUPTS 16
#define MAX_STACK_SIZE 16
#define MAX_QUEUE_SIZE (MAX_PROCESSES + 1)

#define QUANTUM 2
#define MAX_DURATION 10
#define MAX_ARRIVAL_TIME 1000
#define MAX_SIMULATION_TICKS \
    (MAX_ARRIVAL_TIME + (MAX_PROCESSES + MAX_INTERRUPTS) * MAX_DURATION)		#기능 명세
• _CRT_SECURE_NO_WARNINGS: MSVC가 scanf에 내는 보안 경고를 끈다.
• stdio.h: scanf·printf, stdlib.h: 오류 시 종료하는 exit 호출을 위해 포함한다.
• MAX_PROCESSES(8)·MAX_INTERRUPTS(16): 입력 개수 상한, MAX_STACK_SIZE(16): 중단 문맥을 저장할 칸 수
• QUANTUM(2): 한 번 배정받아 연속 실행할 수 있는 최대 tick
• MAX_DURATION(10)·MAX_ARRIVAL_TIME(1000): 실행 시간과 인터럽트 도착 시각의 상한

#분석
• MAX_QUEUE_SIZE = 8 + 1: 원형 큐는 빈 상태와 가득 찬 상태를 구분하려고 한 칸을 비워 두므로, 프로세스 8개를 담으려면 9칸이 필요하다.
• MAX_SIMULATION_TICKS = 1000 + (8 + 16) × 10 = 1240: 마지막 인터럽트가 시각 1000에 와도 남은 일은 전체 작업 시간의 합 240 이하라 정상 입력은 이 안에 끝난다. tick이 이 값에 이르면 main이 상태 전이 오류로 보고 반복을 멈춘다.
typedef enum
{
    IDLE,
    PROCESS,
    INTERRUPT
} TaskType;

typedef struct
{
    TaskType kind;
    int id;
    int remaining;
    int quantum_left;
} Context;

const Context IDLE_CONTEXT = {IDLE, 0, 0, 0};		#기능 명세
• TaskType: CPU가 하는 일의 종류. IDLE(실행할 작업 없음), PROCESS(라운드로빈 대상 프로세스), INTERRUPT(인터럽트 처리)
• Context: 작업 하나의 문맥. kind(종류), id(입력 순서 번호), remaining(남은 실행 시간), quantum_left(이번 배정에서 남은 할당 시간)
• IDLE_CONTEXT: 모든 값이 0인 IDLE 문맥으로, CPU를 비울 때 current에 대입한다.

#분석
• 중단된 작업을 재개하는 데 필요한 값이 이 네 필드뿐이라, 큐와 스택에는 Context를 값 그대로 복사해 넣는다.
• 인터럽트는 라운드로빈 대상이 아니므로 quantum_left를 0으로 둔다.
• IDLE이 열거형의 첫 값(0)이라, 0으로 초기화한 Context는 IDLE_CONTEXT와 같다.
typedef struct
{
    Context queue[MAX_QUEUE_SIZE];
    int front, rear;
} ReadyQueue;

typedef struct
{
    Context stack[MAX_STACK_SIZE];
    int top;
} ContextStack;		#기능 명세
• ReadyQueue: CPU를 기다리는 프로세스의 원형 큐. front는 첫 원소 바로 앞 칸, rear는 마지막 원소가 있는 칸
• ContextStack: 인터럽트로 중단된 문맥의 스택. top은 가장 최근에 넣은 칸의 인덱스이고, 비어 있으면 -1

#분석
• 큐는 선입선출(FIFO)이므로 퀀텀을 다 쓴 프로세스를 뒤에 넣으면 나머지 프로세스가 한 번씩 돈 뒤에 차례가 다시 온다.
• 스택은 후입선출(LIFO)이므로 인터럽트가 겹쳐도 가장 나중에 멈춘 작업부터 재개된다.
typedef struct
{
    ReadyQueue ready;
    ContextStack suspended;
    Context current;
    int tick;
    int finished_processes;
    int finished_interrupts;
} Scheduler;

typedef struct
{
    int arrival_time;
    int id;
    int duration;
} InterruptEvent;		#기능 명세
• Scheduler: 시뮬레이션 상태 전체. ready(준비 큐), suspended(중단 문맥 스택), current(실행 중인 문맥), tick(현재 시각), finished_processes·finished_interrupts(완료 수, main의 반복 종료 조건)
• InterruptEvent: 미리 입력받은 인터럽트 정보. arrival_time(도착 시각), id, duration(처리 시간)

#분석
• current는 포인터가 아닌 값이므로, 큐에서 꺼내거나 스택에서 복원한 문맥은 current로 복사된다.
• 도착 시각이 되면 start_interrupt가 InterruptEvent로 INTERRUPT 문맥을 만든다.
int enqueue(ReadyQueue *q, Context value);
Context dequeue(ReadyQueue *q);
int push(ContextStack *s, Context value);
Context pop(ContextStack *s);

void select_process(Scheduler *s);

• 앞의 5개는 제공 함수 중 큐·스택 조작 함수와 1 tick 실행 함수, 뒤의 3개는 구현 함수의 원형이다.

#분석
• 원형을 앞에 두어 정의 순서와 관계없이 서로 호출할 수 있다.
• 과제 조건대로 함수 형식·매크로·구조체는 수정하지 않았고 함수를 새로 추가하지 않았다.

init_queue

소스코드기능 명세와 분석
void init_queue(ReadyQueue *q)
{
    q->front = q->rear = 0;
}		#기능 명세
• front = rear = 0으로 빈 큐를 만든다. 이 큐에서는 front == rear가 빈 상태다.

is_empty_queue

소스코드기능 명세와 분석
int is_empty_queue(const ReadyQueue *q)
{
    return q->front == q->rear;
}		#기능 명세
• front == rear이면 꺼낼 원소가 없으므로 1, 아니면 0을 반환한다.

is_full_queue

소스코드기능 명세와 분석
int is_full_queue(const ReadyQueue *q)
{
    return (q->rear + 1) % MAX_QUEUE_SIZE == q->front;
}		#기능 명세
• rear의 다음 칸이 front이면 가득 찬 것으로 보고 1을 반환한다.

#분석
• 9칸을 모두 채우면 rear가 front와 겹쳐 빈 큐와 구별되지 않으므로 8칸까지만 쓴다.
• % MAX_QUEUE_SIZE는 마지막 칸 다음을 0번 칸으로 되돌린다.

enqueue

소스코드기능 명세와 분석
int enqueue(ReadyQueue *q, Context value)
{
    if (q == NULL || is_full_queue(q))
    {
        return 0;
    }
    q->rear = (q->rear + 1) % MAX_QUEUE_SIZE;
    q->queue[q->rear] = value;
    return 1;
}		#기능 명세
• 인자: 큐 q, 넣을 문맥 value
• 반환값: 성공하면 1, 큐가 NULL이거나 가득 차면 0(상태 변경 없음)
• 상태 변경: rear를 원형으로 한 칸 옮기고 그 칸에 value를 복사

#분석
• 실패 시 처리는 호출한 쪽이 정한다. main의 초기 등록과 finish_tick의 재등록에서 쓰인다.

dequeue

소스코드기능 명세와 분석
Context dequeue(ReadyQueue *q)
{
    if (q == NULL || is_empty_queue(q))
    {
        printf("QUEUE_ERROR\n");
        exit(1);
    }
    q->front = (q->front + 1) % MAX_QUEUE_SIZE;
    return q->queue[q->front];
}		#기능 명세
• 반환값: 맨 앞 문맥
• 상태 변경: front를 원형으로 한 칸 옮긴다. 빈 큐이면 QUEUE_ERROR를 출력하고 exit(1)로 종료

#분석
• 꺼낸 칸의 값은 지우지 않으며, 이후 enqueue가 덮어쓴다.

init_stack

소스코드기능 명세와 분석
void init_stack(ContextStack *s)
{
    s->top = -1;
}		#기능 명세
• top = -1로 빈 스택을 만든다.

is_empty_stack

소스코드기능 명세와 분석
int is_empty_stack(const ContextStack *s)
{
    return s->top == -1;
}		#기능 명세
• top == -1이면 저장된 문맥이 없으므로 1을 반환한다. finish_tick이 복원할 문맥이 있는지 확인할 때 쓴다.

is_full_stack

소스코드기능 명세와 분석
int is_full_stack(const ContextStack *s)
{
    return s->top == MAX_STACK_SIZE - 1;
}		#기능 명세
• top이 마지막 인덱스 15이면 1을 반환한다.

push

소스코드기능 명세와 분석
int push(ContextStack *s, Context value)
{
    if (s == NULL || is_full_stack(s))
    {
        return 0;
    }
    s->stack[++(s->top)] = value;
    return 1;
}		#기능 명세
• 반환값: 성공하면 1, 스택이 NULL이거나 가득 차면 0
• 상태 변경: top을 먼저 1 올리고(++top) 그 칸에 value를 복사

#분석
• start_interrupt가 중단되는 작업의 문맥을 저장할 때 쓴다.

pop

소스코드기능 명세와 분석
Context pop(ContextStack *s)
{
    if (s == NULL || is_empty_stack(s))
    {
        printf("STACK_ERROR\n");
        exit(1);
    }
    return s->stack[(s->top)--];
}		#기능 명세
• 반환값: top 칸의 문맥. 반환 후 top을 1 내린다(top--).
• 빈 스택이면 STACK_ERROR를 출력하고 exit(1)로 종료

#분석
• finish_tick이 인터럽트 종료 후 가장 최근에 중단된 문맥을 되돌릴 때 쓴다.

select_process (구현)

소스코드기능 명세와 분석
void select_process(Scheduler *s)
{
    if (s->current.kind != IDLE || is_empty_queue(&s->ready))
    {
        return;
    }
    s->current = dequeue(&s->ready);
    s->current.quantum_left = QUANTUM;
}		#기능 명세
• 인자: 스케줄러 s / 반환값: 없음
• 상태 변경: CPU가 비어 있고 준비 큐가 비어 있지 않으면 맨 앞 프로세스를 current로 꺼내 quantum_left = QUANTUM(2)으로 설정

#분석
• current.kind != IDLE 검사: main이 매 tick 호출하므로, 실행 중인 프로세스나 인터럽트를 덮어쓰지 않게 막는다.
• 빈 큐를 먼저 검사해 dequeue의 QUEUE_ERROR 종료를 피한다. 큐가 비면 current는 IDLE로 남고 execute_tick은 시간만 흘려보낸다.
• 큐로 돌아온 프로세스는 quantum_left가 0이므로 선택될 때마다 퀀텀 2를 새로 받는다.
• 스택에서 복원된 프로세스는 current가 IDLE이 아니어서 이 함수를 거치지 않으므로, 중단 전 quantum_left를 이어 쓴다.

start_interrupt (구현)

소스코드기능 명세와 분석
void start_interrupt(Scheduler *s, InterruptEvent event)
{
    if (s->current.kind != IDLE && !push(&s->suspended, s->current))
    {
        printf("SIMULATION_ERROR\n");
        exit(2);
    }
    s->current.kind = INTERRUPT;
    s->current.id = event.id;
    s->current.remaining = event.duration;
    s->current.quantum_left = 0;
}		#기능 명세
• 인자: 스케줄러 s, 도착한 인터럽트 event / 반환값: 없음
• 상태 변경: current가 IDLE이 아니면 스택에 push한 뒤, current를 새 인터럽트(id, remaining = duration, quantum_left = 0)로 바꾼다.

#분석
• 문맥 전환(Context Switch)의 저장 단계다. 중단된 프로세스나 인터럽트는 remaining과 quantum_left가 그대로 저장되어, 복원 후 그 상태에서 이어서 실행된다.
• IDLE은 돌아갈 작업이 아니라 저장하지 않는다. &&의 단락 평가로 current가 IDLE이면 push가 호출되지 않는다.
• push 실패를 무시하면 문맥이 사라져 정상 종료할 수 없으므로 SIMULATION_ERROR를 출력하고 exit(2)로 끝낸다. 정상 입력에서는 스택에 최대 16개(인터럽트 15개 + 프로세스 1개)만 쌓여 이 분기에 오지 않는다.
• execute_tick은 인터럽트의 quantum_left를 검사하거나 줄이지 않으므로, 인터럽트는 퀀텀 소진으로 중단되지 않는다.

finish_tick (구현)

소스코드기능 명세와 분석
void finish_tick(Scheduler *s)
{
    Context *current = &s->current;
    if (current->kind == PROCESS)
    {
        if (current->remaining == 0)
        {
            s->finished_processes++;
            *current = IDLE_CONTEXT;
        }
        else if (current->quantum_left == 0)
        {
            if (!enqueue(&s->ready, *current))
            {
                printf("SIMULATION_ERROR\n");
                exit(2);
            }
            *current = IDLE_CONTEXT;
        }
    }		#기능 명세
• 인자: 스케줄러 s / 반환값: 없음
• 상태 변경: execute_tick 직후 결과에 따라 완료 처리, 큐 재등록, 문맥 복원을 하고 current를 정리한다.
• 지역 변수 current는 s->current를 가리키는 포인터라, 변경이 스케줄러에 바로 반영된다.

#분석
• 프로세스의 remaining == 0: 완료 수를 늘리고 IDLE로 비워, 다음 tick에 select_process가 다른 프로세스를 고르게 한다.
• quantum_left == 0: 퀀텀 소진이므로 준비 큐 맨 뒤에 넣고 IDLE로 비운다. 둘 다 아니면 다음 tick에도 같은 프로세스가 실행된다.
• 완료 검사를 먼저 하는 이유: 두 값이 같은 tick에 함께 0이 될 수 있다(추가 사례 1·2). 순서가 반대면 끝난 프로세스가 큐로 돌아가고, 다시 선택되면 execute_tick이 remaining 0을 보고 SIMULATION_ERROR로 끝난다.
• 실행 중인 프로세스는 큐 밖에 있어 큐에는 최대 7개뿐이므로 정상 입력에서 enqueue는 실패하지 않는다. 실패 시 처리는 start_interrupt와 같다.
    else if (current->kind == INTERRUPT && current->remaining == 0)
    {
        s->finished_interrupts++;
        if (is_empty_stack(&s->suspended))
        {
            *current = IDLE_CONTEXT;
        }
        else
        {
            *current = pop(&s->suspended);
        }
    }
}		#분석
• 인터럽트의 remaining == 0: 완료 수를 늘리고, 스택이 비었으면 IDLE로, 아니면 pop으로 가장 최근에 중단된 문맥을 current로 복원한다.
• 인터럽트가 겹쳤다면 바깥쪽 인터럽트가 먼저 복원되고, 이들이 모두 끝나야 처음 중단된 프로세스가 나온다.
• 빈 스택을 먼저 확인하므로 pop의 STACK_ERROR 종료는 일어나지 않는다. 다음 프로세스 선택은 select_process가 맡는다.

execute_tick

소스코드기능 명세와 분석
int execute_tick(Scheduler *s)
{
    Context *current = &s->current;
    if ((current->kind != IDLE && current->remaining <= 0) ||
        (current->kind == PROCESS && current->quantum_left <= 0))
    {
        return 0;
    }		#기능 명세
• 인자: 스케줄러 s
• 반환값: 1 tick을 실행했으면 1, 실행할 수 없는 상태면 0

#분석
• IDLE이 아닌 작업의 remaining이 0 이하이거나 프로세스의 quantum_left가 0 이하이면 즉시 0을 반환한다(조기 반환, Early Return).
• finish_tick이 상태를 올바르게 정리했다면 나올 수 없는 상태라, 구현 함수의 상태 전이를 점검하는 역할을 한다. main은 0을 받으면 SIMULATION_ERROR로 끝낸다.
    if (current->kind != IDLE)
    {
        current->remaining--;
    }
    s->tick++;
    if (current->kind == PROCESS)
    {
        current->quantum_left--;
        printf("T%3d P%2d R%2d Q%2d\n", s->tick,
               current->id, current->remaining, current->quantum_left);
    }
    else if (current->kind == INTERRUPT)
    {
        printf("T%3d I%2d R%2d\n", s->tick, current->id, current->remaining);
    }
    else
    {
        printf("T%3d IDLE\n", s->tick);
    }
    return 1;
}		#기능 명세
• 상태 변경: IDLE이 아니면 remaining을 1 줄이고, 프로세스면 quantum_left도 1 줄인다. tick은 항상 1 늘린다.
• 출력: 프로세스는 T·P·R·Q, 인터럽트는 T·I·R, IDLE은 T와 IDLE

#분석
• T는 tick을 늘린 뒤의 시각이므로, T 1은 시각 0부터 1까지의 실행 결과다.
• 완료·재등록·복원은 출력 뒤 finish_tick이 맡는다.

실행결과 및 분석

공식 예시: 중첩 인터럽트와 라운드로빈

입력:

2 2
3
1
1 2
2 1

실제 출력:

T  1 P 1 R 2 Q 1
T  2 I 1 R 1
T  3 I 2 R 0
T  4 I 1 R 0
T  5 P 1 R 1 Q 0
T  6 P 2 R 0 Q 1
T  7 P 1 R 0 Q 1
FINISHED 2 INTERRUPTS 2 TICKS 7
시각실행상태 변화의 이유
0→1P1준비 큐 [P1, P2]에서 P1을 꺼내 Q=2를 준다. 1 tick 실행 후 R2 Q1, 두 값 모두 0이 아니므로 계속 실행한다.
1→2I1I1 도착. P1(R2, Q1)을 push해 스택 [P1]. I1 R2→1.
2→3I2I2 도착. I1(R1)을 push해 스택 [P1, I1]. I2 R1→0으로 완료, 맨 위 I1을 pop한다.
3→4I1I1 R1→0으로 완료, 스택에 남은 P1을 pop해 스택이 빈다.
4→5P1복원된 P1은 Q1을 이어 써서 R1 Q0. 퀀텀 소진으로 큐 뒤에 넣어 [P2, P1], CPU는 IDLE.
5→6P2P2를 꺼내 Q=2. R1→0으로 완료.
6→7P1P1을 다시 꺼내 Q=2. R1→0으로 완료. 모든 작업이 끝나 반복 종료.

결과 분석: 출력 순서 P1 I1 I2 I1 P1 P2 P1은 과제 18쪽 실행 화면·간트 차트와 같다. 시각 2에 I2가 I1을 선점(Preemption)했고, I2가 끝나자 처음 멈춘 P1이 아니라 가장 최근에 멈춘 I1이 먼저 재개되었다. T5의 Q 0은 T1에서 남긴 퀀텀 1을 복원 후 이어 쓴 결과이고, T7의 Q 1은 큐에서 다시 선택되며 퀀텀 2를 새로 받은 결과다.

추가 사례 1: 세 겹 중첩 인터럽트

입력:

1 3
4
1 3
2 3
3 1

실제 출력:

T  1 P 1 R 3 Q 1
T  2 I 1 R 2
T  3 I 2 R 2
T  4 I 3 R 0
T  5 I 2 R 1
T  6 I 2 R 0
T  7 I 1 R 1
T  8 I 1 R 0
T  9 P 1 R 2 Q 0
T 10 P 1 R 1 Q 1
T 11 P 1 R 0 Q 0
FINISHED 1 INTERRUPTS 3 TICKS 11
시각실행상태 변화의 이유
0→1P1P1 선택, Q=2. R3 Q1.
1→4I1, I2, I3시각 1·2·3에 차례로 도착하며 P1, I1, I2를 push. 스택 [P1, I1, I2]. I3은 1 tick에 완료하고 I2를 pop.
4→6I2I2가 남은 2 tick을 실행해 완료, I1을 pop.
6→8I1I1이 남은 2 tick을 실행해 완료, P1을 pop. 스택이 빈다.
8→9P1Q1을 이어 써서 R2 Q0. 퀀텀 소진으로 큐 [P1]에 넣고 IDLE.
9→11P1큐에 P1뿐이므로 다시 꺼내 Q=2. T11에 R0과 Q0이 동시에 되고, 완료가 먼저 검사되어 완료로 처리.

결과 분석: 인터럽트는 I1, I2, I3 순으로 시작해 I3, I2, I1 순으로 끝났다. 스택에 늦게 들어간 문맥일수록 먼저 복원되어(최대 3개 적재) P1이 가장 늦게 재개되었다. T11에는 R과 Q가 동시에 0이 되었지만 완료로 처리되어 FINISHED 1이 출력되었다.

추가 사례 2: 완료와 퀀텀 소진이 같은 tick

입력:

2 0
2
1

실제 출력:

T  1 P 1 R 1 Q 1
T  2 P 1 R 0 Q 0
T  3 P 2 R 0 Q 1
FINISHED 2 INTERRUPTS 0 TICKS 3
시각실행상태 변화의 이유
0→2P1Q=2로 2 tick 실행. T2에 R0 Q0. finish_tick이 remaining을 먼저 보고 완료 처리, IDLE.
2→3P2큐에 남은 P2를 꺼내 1 tick에 완료.

결과 분석: P1은 T2에 퀀텀과 남은 시간이 함께 0이 되어 준비 큐로 돌아가지 않고 완료되었다. 그래서 다음 tick에 바로 P2가 실행되었고 P1은 다시 선택되지 않았다.

추가 사례 3: CPU가 빈 순간에 도착한 인터럽트

입력:

2 1
3
2
2 1

실제 출력:

T  1 P 1 R 2 Q 1
T  2 P 1 R 1 Q 0
T  3 I 1 R 0
T  4 P 2 R 1 Q 1
T  5 P 2 R 0 Q 0
T  6 P 1 R 0 Q 1
FINISHED 2 INTERRUPTS 1 TICKS 6
시각실행상태 변화의 이유
0→2P1P1이 퀀텀 2를 모두 써서 R1 Q0. 큐 뒤에 넣어 [P2, P1], IDLE.
2→3I1I1 도착. current가 IDLE이므로 push하지 않음. select_process는 current가 인터럽트라 P2를 꺼내지 않는다. I1 완료 후 스택이 비어 IDLE.
3→5P2P2를 꺼내 Q=2로 2 tick 실행, T5에 R0 Q0으로 완료.
5→6P1P1을 꺼내 1 tick에 완료.

결과 분석: 시각 2에는 P1이 이미 큐로 돌아가 CPU가 비어 있었으므로 I1은 아무 문맥도 push하지 않았고, I1이 끝나자 CPU는 IDLE이 되었다. CPU가 빈 시각에도 main이 인터럽트를 먼저 시작하므로 T3에는 P2가 아닌 I1이 실행되었고, 이후 큐 순서 [P2, P1]대로 실행되었다.

추가 사례 4: 프로세스가 모두 끝난 뒤의 인터럽트

입력:

1 1
1
3 2

실제 출력:

T  1 P 1 R 0 Q 1
T  2 IDLE
T  3 IDLE
T  4 I 1 R 1
T  5 I 1 R 0
FINISHED 1 INTERRUPTS 1 TICKS 5
시각실행상태 변화의 이유
0→1P1P1이 1 tick에 완료, IDLE.
1→3IDLE인터럽트 완료 수가 0이라 반복이 계속된다. 큐가 비어 select_process가 아무것도 고르지 않아 시간만 흐른다.
3→5I1시각 3에 I1 도착, IDLE이므로 push 없음. 2 tick 처리 후 스택이 비어 IDLE로 끝난다.

결과 분석: P1은 시각 1에 끝났지만 main의 반복 조건이 인터럽트 완료 수까지 보므로 프로그램이 계속되었다. 시각 1~3에는 실행할 작업이 없어 IDLE이 두 번 출력되었고, 이 2 tick도 TICKS 5에 포함된다.

전체 소스코드

#define _CRT_SECURE_NO_WARNINGS

#include <stdio.h>
#include <stdlib.h>

#define MAX_PROCESSES 8
#define MAX_INTERRUPTS 16
#define MAX_STACK_SIZE 16
#define MAX_QUEUE_SIZE (MAX_PROCESSES + 1)

#define QUANTUM 2
#define MAX_DURATION 10
#define MAX_ARRIVAL_TIME 1000
#define MAX_SIMULATION_TICKS \
    (MAX_ARRIVAL_TIME + (MAX_PROCESSES + MAX_INTERRUPTS) * MAX_DURATION)

typedef enum
{
    IDLE,
    PROCESS,
    INTERRUPT
} TaskType;

typedef struct
{
    TaskType kind;
    int id;
    int remaining;
    int quantum_left;
} Context;

const Context IDLE_CONTEXT = {IDLE, 0, 0, 0};

typedef struct
{
    Context queue[MAX_QUEUE_SIZE];
    int front, rear;
} ReadyQueue;

typedef struct
{
    Context stack[MAX_STACK_SIZE];
    int top;
} ContextStack;

typedef struct
{
    ReadyQueue ready;
    ContextStack suspended;
    Context current;
    int tick;
    int finished_processes;
    int finished_interrupts;
} Scheduler;

typedef struct
{
    int arrival_time;
    int id;
    int duration;
} InterruptEvent;

int enqueue(ReadyQueue *q, Context value);
Context dequeue(ReadyQueue *q);
int push(ContextStack *s, Context value);
Context pop(ContextStack *s);
int execute_tick(Scheduler *s);

void select_process(Scheduler *s);
void start_interrupt(Scheduler *s, InterruptEvent event);
void finish_tick(Scheduler *s);

void init_queue(ReadyQueue *q)
{
    q->front = q->rear = 0;
}

int is_empty_queue(const ReadyQueue *q)
{
    return q->front == q->rear;
}

int is_full_queue(const ReadyQueue *q)
{
    return (q->rear + 1) % MAX_QUEUE_SIZE == q->front;
}

int enqueue(ReadyQueue *q, Context value)
{
    if (q == NULL || is_full_queue(q))
    {
        return 0;
    }
    q->rear = (q->rear + 1) % MAX_QUEUE_SIZE;
    q->queue[q->rear] = value;
    return 1;
}

Context dequeue(ReadyQueue *q)
{
    if (q == NULL || is_empty_queue(q))
    {
        printf("QUEUE_ERROR\n");
        exit(1);
    }
    q->front = (q->front + 1) % MAX_QUEUE_SIZE;
    return q->queue[q->front];
}

void init_stack(ContextStack *s)
{
    s->top = -1;
}

int is_empty_stack(const ContextStack *s)
{
    return s->top == -1;
}

int is_full_stack(const ContextStack *s)
{
    return s->top == MAX_STACK_SIZE - 1;
}

int push(ContextStack *s, Context value)
{
    if (s == NULL || is_full_stack(s))
    {
        return 0;
    }
    s->stack[++(s->top)] = value;
    return 1;
}

Context pop(ContextStack *s)
{
    if (s == NULL || is_empty_stack(s))
    {
        printf("STACK_ERROR\n");
        exit(1);
    }
    return s->stack[(s->top)--];
}

void select_process(Scheduler *s)
{
    if (s->current.kind != IDLE || is_empty_queue(&s->ready))
    {
        return;
    }
    s->current = dequeue(&s->ready);
    s->current.quantum_left = QUANTUM;
}

void start_interrupt(Scheduler *s, InterruptEvent event)
{
    if (s->current.kind != IDLE && !push(&s->suspended, s->current))
    {
        printf("SIMULATION_ERROR\n");
        exit(2);
    }
    s->current.kind = INTERRUPT;
    s->current.id = event.id;
    s->current.remaining = event.duration;
    s->current.quantum_left = 0;
}

void finish_tick(Scheduler *s)
{
    Context *current = &s->current;
    if (current->kind == PROCESS)
    {
        if (current->remaining == 0)
        {
            s->finished_processes++;
            *current = IDLE_CONTEXT;
        }
        else if (current->quantum_left == 0)
        {
            if (!enqueue(&s->ready, *current))
            {
                printf("SIMULATION_ERROR\n");
                exit(2);
            }
            *current = IDLE_CONTEXT;
        }
    }
    else if (current->kind == INTERRUPT && current->remaining == 0)
    {
        s->finished_interrupts++;
        if (is_empty_stack(&s->suspended))
        {
            *current = IDLE_CONTEXT;
        }
        else
        {
            *current = pop(&s->suspended);
        }
    }
}

int execute_tick(Scheduler *s)
{
    Context *current = &s->current;
    if ((current->kind != IDLE && current->remaining <= 0) ||
        (current->kind == PROCESS && current->quantum_left <= 0))
    {
        return 0;
    }

    if (current->kind != IDLE)
    {
        current->remaining--;
    }
    s->tick++;
    if (current->kind == PROCESS)
    {
        current->quantum_left--;
        printf("T%3d P%2d R%2d Q%2d\n", s->tick,
               current->id, current->remaining, current->quantum_left);
    }
    else if (current->kind == INTERRUPT)
    {
        printf("T%3d I%2d R%2d\n", s->tick, current->id, current->remaining);
    }
    else
    {
        printf("T%3d IDLE\n", s->tick);
    }
    return 1;
}

int main(void)
{
    Context processes[MAX_PROCESSES] = {0};
    InterruptEvent interrupts[MAX_INTERRUPTS] = {0};
    int process_count, interrupt_count;
    Scheduler s = {0};
    int next_interrupt = 0;

    if (scanf("%d %d", &process_count, &interrupt_count) != 2 ||
        process_count < 1 || process_count > MAX_PROCESSES ||
        interrupt_count < 0 || interrupt_count > MAX_INTERRUPTS)
    {
        printf("INPUT_ERROR\n");
        return 1;
    }

    for (int i = 0; i < process_count; i++)
    {
        if (scanf("%d", &processes[i].remaining) != 1 ||
            processes[i].remaining < 1 || processes[i].remaining > MAX_DURATION)
        {
            printf("INPUT_ERROR\n");
            return 1;
        }
        processes[i].id = i + 1;
        processes[i].kind = PROCESS;
    }

    for (int i = 0; i < interrupt_count; i++)
    {
        if (scanf("%d %d", &interrupts[i].arrival_time,
                  &interrupts[i].duration) != 2 ||
            interrupts[i].arrival_time < 0 || interrupts[i].arrival_time > MAX_ARRIVAL_TIME ||
            interrupts[i].duration < 1 || interrupts[i].duration > MAX_DURATION ||
            (i > 0 && interrupts[i].arrival_time <= interrupts[i - 1].arrival_time))
        {
            printf("INPUT_ERROR\n");
            return 1;
        }
        interrupts[i].id = i + 1;
    }

    init_queue(&s.ready);
    init_stack(&s.suspended);

    for (int i = 0; i < process_count; i++)
    {
        if (!enqueue(&s.ready, processes[i]))
        {
            printf("SIMULATION_ERROR\n");
            return 2;
        }
    }

    while (s.finished_processes < process_count ||
           s.finished_interrupts < interrupt_count)
    {
        if (s.tick >= MAX_SIMULATION_TICKS)
        {
            printf("SIMULATION_ERROR\n");
            return 2;
        }

        if (next_interrupt < interrupt_count &&
            interrupts[next_interrupt].arrival_time == s.tick)
        {
            start_interrupt(&s, interrupts[next_interrupt]);
            next_interrupt++;
        }
        select_process(&s);
        if (!execute_tick(&s))
        {
            printf("SIMULATION_ERROR\n");
            return 2;
        }
        finish_tick(&s);
    }

    printf("FINISHED %d INTERRUPTS %d TICKS %d\n",
           s.finished_processes, s.finished_interrupts, s.tick);
    return 0;
}
