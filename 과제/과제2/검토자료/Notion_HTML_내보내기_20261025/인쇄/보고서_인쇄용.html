구현 목적

메모리를 원형 이중 연결 리스트로 표현하며 각 노드는 한 구간을 기록한다. 할당 시 First-Fit으로 요청 크기 이상인 첫 빈 블록을 찾는다. 공간이 더 크면 할당 부분과 남는 부분으로 나눈다. 해제 시 FREE로 바꾸고 인접한 빈 블록과 합친다. 실패하면 기존 메모리 지도를 유지한다.

헤더·상수·구조체·함수 원형

소스코드기능 명세와 분석
#define _CRT_SECURE_NO_WARNINGS

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define FREE_PID (-1)
#define HEAD_PID (-2)
#define MAX_COMMANDS 10000
#define COMMAND_LENGTH 16

stdio.h는 scanf와 printf를 사용해 입력을 받고 결과를 출력하기 위해 필요하다. stdlib.h는 malloc과 free로 노드를 생성하고 해제하는 데 필요하고, string.h는 strcmp로 입력 명령이 ALLOC인지 FREE인지 비교하는 데 필요하다.

_CRT_SECURE_NO_WARNINGS는 MSVC의 관련 보안 경고를 억제하는 매크로이며 입력을 검증하거나 안전성을 보장하는 기능은 아니다.

FREE_PID(-1)는 빈 블록, HEAD_PID(-2)는 헤드의 표식이다. 실제 PID는 양수다. MAX_COMMANDS(10000)는 명령 개수 상한이고, COMMAND_LENGTH(16)는 명령 문자열의 저장 공간이다. scanf의 %15s와 결합해 널 문자 한 칸을 남긴다.
typedef struct MemoryBlock
{
    int start;
    int size;
    int pid;
    struct MemoryBlock *prev;
    struct MemoryBlock *next;
} MemoryBlock;

제공된 MemoryBlock에서 start는 블록의 시작 주소, size는 크기, pid는 사용 중인 프로세스 번호나 FREE 상태를 나타낸다. prev와 next는 이전·다음 노드를 가리켜 리스트를 양방향으로 연결한다. 끝 주소는 start + size - 1이다.

노드의 앞뒤 순서는 주소 순서와 일치한다. 빈 구간도 FREE_PID를 가진 노드로 기록하므로 사용 중인 구간과 빈 구간을 모두 리스트에서 추적할 수 있다.
typedef struct
{
    MemoryBlock head;
    int total_size;
} MemoryList;

제공된 MemoryList는 기준 노드 head 자체와 전체 크기 total_size를 포함한다. head는 실제 할당 대상이 아니고 원형 리스트의 시작과 끝을 구분한다.

실제 노드가 하나도 없을 때 head.prev와 head.next는 head 자신을 가리킨다. 별도의 파일 범위 전역 상태 변수는 없으며 main의 지역 변수 memory를 포인터로 전달해 상태를 공유한다.

앞의 6개는 리스트 생성·삽입·삭제·정리·출력을 담당하는 제공 함수다. 뒤의 5개는 탐색·할당·병합·반납을 담당하는 구현 대상 함수다. 선언한 함수 형식과 자료구조는 유지하고 추가 함수는 만들지 않았다.

리스트 포인터를 받은 함수가 같은 상태를 참조한다. print_memory_map의 const는 출력 함수가 전달받은 리스트를 수정하지 않는 용도를 나타낸다. 각 함수의 실제 변경 상태와 반환값은 다음 항목에서 구분한다.

init_list

제공 함수

소스코드기능 명세와 분석
int init_list(MemoryList *list, int total_size)
{
    MemoryBlock *initial_block;

    if (list == NULL || total_size <= 0)
    {
        return 0;
    }

    list->head.start = -1;
    list->head.size = 0;
    list->head.pid = HEAD_PID;
    list->head.prev = &list->head;
    list->head.next = &list->head;
    list->total_size = total_size;

    initial_block = create_block(0, total_size, FREE_PID);
    if (initial_block == NULL)
    {
        list->total_size = 0;
        return 0;
    }

    insert_after(&list->head, initial_block);
    return 1;
}		인자: 초기화할 MemoryList 포인터와 전체 크기. 반환값: 성공 1, 잘못된 인자나 초기 노드 생성 실패 시 0. 변경 상태: 헤드의 연결, total_size, 초기 FREE 블록.

헤드는 실제 주소 구간을 나타내지 않는 기준 노드다. 먼저 prev와 next를 자기 자신으로 설정해 빈 원형 리스트를 만든다. 아직 어떤 프로세스에도 할당하지 않았으므로 전체 메모리가 하나의 연속된 빈 공간이다. 따라서 시작 주소 0, 크기 total_size인 FREE 블록 하나로 표현한다.

초기 블록을 만들지 못하면 total_size를 0으로 되돌리고 실패를 반환한다. 헤드만 자기 자신을 가리키는 상태이므로 불완전한 실제 블록이 리스트에 남지 않는다.

create_block

제공 함수

소스코드기능 명세와 분석
MemoryBlock *create_block(int start, int size, int pid)
{
    MemoryBlock *block;

    block = (MemoryBlock *)malloc(sizeof(MemoryBlock));
    if (block == NULL)
    {
        return NULL;
    }

    block->start = start;
    block->size = size;
    block->pid = pid;
    block->prev = NULL;
    block->next = NULL;
    return block;
}		인자: 시작 주소, 크기, PID. 반환값: 생성한 MemoryBlock 포인터 또는 NULL. 변경 상태: 성공 시 노드 한 개를 동적 생성한다. 리스트 연결은 변경하지 않는다.

malloc으로 확보하는 것은 프로세스가 사용할 size칸의 실제 RAM이 아니라, start·size·pid와 링크를 보관하는 관리 노드의 저장 공간이다. 따라서 요청량과 관계없이 sizeof(MemoryBlock)만큼 생성한다.

malloc이 메모리 확보에 실패하면 NULL을 반환하므로 먼저 확인해야 한다. NULL인 상태에서 노드의 필드에 접근하면 유효하지 않은 메모리에 접근하는 문제가 생기며, C에서는 정의되지 않은 동작이다. 생성 성공 후 prev와 next는 NULL로 초기화하고, 실제 리스트 연결은 insert_after가 담당한다.

insert_after

제공 함수

소스코드기능 명세와 분석
void insert_after(MemoryBlock *position, MemoryBlock *new_block)
{
    if (position == NULL || new_block == NULL)
    {
        return;
    }

    new_block->prev = position;
    new_block->next = position->next;
    position->next->prev = new_block;
    position->next = new_block;
}		인자: 삽입 기준 노드 position과 새 노드 new_block. 반환값: 없음. 변경 상태: 두 노드와 기존 다음 노드의 prev·next 연결.

position과 원래 다음 노드 사이에 새 노드를 넣는다. 새 노드의 prev는 position, next는 원래 다음 노드를 가리키게 한다. 이어서 원래 다음 노드의 prev와 position의 next를 새 노드로 바꾼다.

이중 연결 리스트는 앞뒤로 이동하므로 next뿐 아니라 prev도 새 연결에 맞게 바꿔야 한다. prev를 갱신하지 않으면 뒤로 이동할 때 새 노드를 건너뛰거나 잘못된 노드를 가리킬 수 있다. position->next를 마지막에 바꾸므로 원래 다음 노드를 잃지 않는다.

remove_block

제공 함수

소스코드기능 명세와 분석
void remove_block(MemoryBlock *removed)
{
    if (removed == NULL || removed->pid == HEAD_PID ||
        removed->prev == NULL || removed->next == NULL)
    {
        return;
    }

    removed->prev->next = removed->next;
    removed->next->prev = removed->prev;
    removed->prev = NULL;
    removed->next = NULL;
    free(removed);
}		인자: 제거할 노드 removed. 반환값: 없음. 변경 상태: 양옆 노드를 직접 연결하고 removed의 동적 저장 공간을 해제한다.

NULL, 헤드 PID, 연결되지 않은 노드는 제거하지 않는다. 정상 노드는 이전 노드의 next와 다음 노드의 prev를 서로 연결한 뒤 free한다. 헤드가 있어서 첫 블록의 prev와 마지막 블록의 next도 NULL이 아니라 헤드를 가리킨다. 그래서 처음이나 끝의 블록을 삭제할 때도 양옆 노드를 연결하는 같은 코드로 처리할 수 있다.

이 함수 호출 후 기존 removed 포인터로 노드 멤버를 읽으면 안 된다. 또한 이 함수 자체는 가상 주소 구간을 이웃 블록에 합치지 않는다. 병합 함수가 먼저 크기를 합산한 뒤 불필요해진 관리 노드만 삭제해야 전체 구간이 보존된다.

제공 함수

소스코드기능 명세와 분석
void destroy_list(MemoryList *list)
{
    if (list == NULL)
    {
        return;
    }

    while (list->head.next != &list->head)
    {
        remove_block(list->head.next);
    }

    list->head.prev = &list->head;
    list->head.next = &list->head;
    list->total_size = 0;
}		인자: 정리할 리스트. 반환값: 없음. 변경 상태: 모든 실제 노드를 해제하고 헤드 연결과 total_size를 초기 상태로 정리한다.

헤드 다음 노드가 헤드 자체가 될 때까지 첫 실제 노드를 반복해서 제거한다. 삭제된 노드의 next를 따라가지 않고 매번 list->head.next를 다시 읽기 때문에 해제된 노드를 다음 반복에서 참조하지 않는다.

헤드는 MemoryList 내부에 포함된 객체이므로 free하지 않는다. 실제 노드를 모두 없앤 뒤 prev·next를 헤드 자신으로 맞추고 total_size를 0으로 설정한다. 노드 수를 n이라 하면 전체 정리에는 O(n)이 필요하다.

제공 함수

소스코드기능 명세와 분석
void print_memory_map(const MemoryList *list)
{
    const MemoryBlock *block;

    if (list == NULL)
    {
        return;
    }

    printf("Memory map: ");
    block = list->head.next;
    while (block != &list->head)
    {
        printf("[%d-%d, ", block->start, block->start + block->size - 1);
        if (block->pid == FREE_PID)
        {
            printf("FREE]");
        }
        else
        {
            printf("P%d]", block->pid);
        }

        block = block->next;
        if (block != &list->head)
        {
            printf(" <-> ");
        }
    }
    printf("\n");
}		인자: 읽기 전용 리스트 포인터. 반환값: 없음. 변경 상태: 리스트를 변경하지 않고 표준 출력에 메모리 지도를 쓴다.

헤드 다음부터 실제 블록을 순회한다. 시작 주소 20에서 크기가 10이면 20~29까지 사용한다. 시작 주소도 한 칸에 포함되므로 마지막 주소는 start + size가 아니라 start + size - 1이다.

FREE_PID이면 FREE, 아니면 P와 PID를 출력한다. 다음 노드가 헤드가 아닐 때만 연결 표시를 넣어 마지막에 불필요한 구분자가 남지 않게 한다. 탐색 중 노드가 헤드로 돌아오면 한 바퀴를 마친 것이다. 노드 수에 비례해 O(n)의 순회가 필요하다.

find_first_fit

구현 대상 함수

소스코드기능 명세와 분석
MemoryBlock *find_first_fit(MemoryList *list, int size)
{
    MemoryBlock *current;

    if (list == NULL || size <= 0)
    {
        return NULL;
    }

    current = list->head.next;
    while (current != &list->head)
    {
        if (current->pid == FREE_PID && current->size >= size)
        {
            return current;
        }
        current = current->next;
    }
    return NULL;
}		인자: 리스트와 요청 크기. 반환값: 조건을 만족하는 첫 FREE 블록의 포인터, 없거나 인자가 잘못되면 NULL. 변경 상태: 없음.

head.next부터 시작해서 헤드로 돌아올 때까지 탐색한다. pid가 FREE_PID이고 크기가 요청 크기 이상이면 그 블록을 바로 반환하고, 아니면 next로 이동한다. 끝까지 없으면 NULL을 반환한다. 조건을 만족하는 즉시 반환하므로 주소 순서상 첫 적합 블록이 선택된다.

원형 리스트에서는 마지막 노드의 next가 NULL이 아니라 헤드다. 따라서 current != &list->head로 종료하며 헤드를 실제 후보로 검사하지 않는다. 최악에는 모든 실제 블록을 확인하므로 O(n)이다. 크기 15와 10의 빈 블록이 이 순서로 있고 8칸을 요청하면 앞의 15칸을 선택한다.

find_process_block

구현 대상 함수

소스코드기능 명세와 분석
MemoryBlock *find_process_block(MemoryList *list, int pid)
{
    MemoryBlock *current;

    if (list == NULL || pid <= 0)
    {
        return NULL;
    }

    current = list->head.next;
    while (current != &list->head)
    {
        if (current->pid == pid)
        {
            return current;
        }
        current = current->next;
    }
    return NULL;
}		인자: 리스트와 양의 PID. 반환값: 해당 PID를 가진 블록 포인터, 없거나 인자가 잘못되면 NULL. 변경 상태: 없음.

헤드 다음부터 한 바퀴 동안 pid가 입력과 같은 노드를 탐색한다. PID가 양수인 경우만 찾으므로 FREE_PID(-1)와 HEAD_PID(-2)를 프로세스로 반환하지 않는다.

할당에서는 같은 PID가 이미 존재하는지 확인하는 데, 반납에서는 반납할 블록을 찾는 데 재사용한다. 현재 구현은 PID 한 개에 블록 한 개를 대응시키며 중복 할당을 거부한다. 최악의 탐색 시간은 O(n)이다.

allocate_memory

구현 대상 함수

소스코드기능 명세와 분석
int allocate_memory(MemoryList *list, int pid, int size)
{
    if (list == NULL || pid <= 0 || size <= 0)
    {
        return 0;
    }
    if (find_process_block(list, pid) != NULL)
    {
        return 0;
    }

    MemoryBlock *block = find_first_fit(list, size);
    if (block == NULL)
    {
        return 0;
    }
    if (block->size == size)
    {
        block->pid = pid;
        return 1;
    }

인자: 리스트, PID, 요청 크기. 반환값: 성공 1, 실패 0. 변경 상태: 성공 시 선택한 블록의 소유자와 필요에 따른 크기·리스트 연결. 실패 시 기존 지도 유지.

먼저 잘못된 입력, 중복 PID, 적합한 빈 블록의 부재를 검사한다. 이때는 기존 블록을 변경하지 않는다. 찾은 블록 크기와 요청 크기가 같으면 분할 없이 PID만 바꾸고 1을 반환한다. 중복 PID 검사와 First-Fit 탐색이 각각 O(n)이지만 순서대로 한 번씩 실행하므로 전체는 O(n)이다.
    MemoryBlock *remaining = create_block(block->start + size,
                                          block->size - size, FREE_PID);
    if (remaining == NULL)
    {
        return 0;
    }
    block->size = size;
    block->pid = pid;
    insert_after(block, remaining);
    return 1;
}		선택한 블록이 더 크면 남는 블록의 시작 주소는 block->start + size, 크기는 block->size - size다. 새 노드 생성에 실패하면 기존 메모리 지도를 그대로 유지해야 한다. 기존 블록을 먼저 바꾸면 남는 공간을 기록할 노드가 없는 상태가 될 수 있으므로, 새 노드 생성에 성공한 뒤 크기와 PID를 변경한다.

남은 FREE 노드는 insert_after로 할당 블록 바로 뒤에 연결한다. 두 구간의 주소 순서와 원래 크기의 합이 유지되고, 성공 시 1을 반환한다.

merge_free_blocks

구현 대상 함수

소스코드기능 명세와 분석
MemoryBlock *merge_free_blocks(MemoryList *list, MemoryBlock *current)
{
    MemoryBlock *previous;
    MemoryBlock *next;

    if (list == NULL || current == NULL ||
        current == &list->head || current->pid != FREE_PID)
    {
        return current;
    }

    previous = current->prev;
    if (previous != &list->head && previous->pid == FREE_PID)
    {
        previous->size += current->size;
        remove_block(current);
        /* current가 삭제되었으므로 살아 있는 이전 노드를 기준으로 바꾼다. */
        current = previous;
    }

인자: 리스트와 병합 기준 FREE 노드. 반환값: 병합 후 살아 있는 기준 노드. 변경 상태: 인접 FREE 구간의 크기와 링크, 중복 관리 노드의 삭제.

잘못된 인자, 헤드, 사용 중 블록은 병합하지 않는다. 왼쪽이 실제 FREE 블록이면 previous의 크기에 current의 크기를 더한 뒤 current를 삭제한다. remove_block(current)에서 current가 가리키던 노드가 해제되므로, 그대로 current->next에 접근하면 해제된 메모리를 사용하는 문제가 생긴다. 그래서 살아 있는 previous를 current로 지정한 뒤 오른쪽 병합을 계속해야 한다.
    next = current->next;
    if (next != &list->head && next->pid == FREE_PID)
    {
        current->size += next->size;
        remove_block(next);
    }

    return current;
}		왼쪽 병합 후 current->next를 다시 읽어 오른쪽 FREE 블록도 합친다. 헤드는 실제 메모리 구간이 아니라 리스트의 시작과 끝을 표시하는 노드다. 마지막 블록의 next가 head를 가리켜도 실제 주소가 이어지는 것은 아니므로 병합하면 안 된다.

정상적인 할당·반납 뒤에는 인접 FREE 블록이 남지 않는 상태가 유지된다. 이 전제에서 한 블록을 반납할 때 양옆만 검사하면 되므로 이 함수 자체는 O(1)이다. 예를 들어 [0~19 FREE], [20~49 FREE], [50~99 FREE]는 시작 0·크기 100인 한 블록으로 합쳐진다.

release_memory

구현 대상 함수

소스코드기능 명세와 분석
int release_memory(MemoryList *list, int pid)
{
    if (list == NULL || pid <= 0)
    {
        return 0;
    }
    MemoryBlock *block = find_process_block(list, pid);
    if (block == NULL)
    {
        return 0;
    }
    block->pid = FREE_PID;
    merge_free_blocks(list, block);
    return 1;
}		인자: 리스트와 반납할 PID. 반환값: 해당 블록 반납 성공 시 1, 찾지 못하면 0. 변경 상태: 블록을 FREE로 표시하고 가능한 이웃 FREE 블록과 병합한다.

PID를 찾지 못하면 리스트를 변경하지 않고 실패한다. 찾으면 pid를 FREE_PID로 바꾼다. 이는 프로세스에 배정한 가상 구간을 사용 가능한 공간으로 돌려놓는 처리다.

그 노드는 해당 메모리 구간의 시작 주소와 크기를 기록하므로, 삭제하면 빈 공간의 정보가 사라진다. 따라서 노드는 유지하고 pid만 FREE_PID로 바꿔야 나중에 다시 할당할 수 있다. 이후 merge_free_blocks로 이웃과 합칠 때만 중복 노드를 제거한다. 병합 후 block을 다시 사용하지 않으므로 반환 포인터를 저장하지 않아도 된다. PID 탐색 O(n)과 병합 O(1)을 합쳐 최악 시간은 O(n)이다.

실제 터미널 실행 화면

전체 소스코드

#define _CRT_SECURE_NO_WARNINGS

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define FREE_PID (-1)
#define HEAD_PID (-2)
#define MAX_COMMANDS 10000
#define COMMAND_LENGTH 16

typedef struct MemoryBlock
{
    int start;
    int size;
    int pid;
    struct MemoryBlock *prev;
    struct MemoryBlock *next;
} MemoryBlock;

typedef struct
{
    MemoryBlock head;
    int total_size;
} MemoryList;

int init_list(MemoryList *list, int total_size);
MemoryBlock *create_block(int start, int size, int pid);
void insert_after(MemoryBlock *position, MemoryBlock *new_block);
void remove_block(MemoryBlock *removed);
void destroy_list(MemoryList *list);
void print_memory_map(const MemoryList *list);

/* 학생 구현 함수 */
MemoryBlock *find_first_fit(MemoryList *list, int size);
MemoryBlock *find_process_block(MemoryList *list, int pid);
int allocate_memory(MemoryList *list, int pid, int size);
MemoryBlock *merge_free_blocks(MemoryList *list, MemoryBlock *current);
int release_memory(MemoryList *list, int pid);

int init_list(MemoryList *list, int total_size)
{
    MemoryBlock *initial_block;

    if (list == NULL || total_size <= 0)
    {
        return 0;
    }

    list->head.start = -1;
    list->head.size = 0;
    list->head.pid = HEAD_PID;
    list->head.prev = &list->head;
    list->head.next = &list->head;
    list->total_size = total_size;

    initial_block = create_block(0, total_size, FREE_PID);
    if (initial_block == NULL)
    {
        list->total_size = 0;
        return 0;
    }

    insert_after(&list->head, initial_block);
    return 1;
}

MemoryBlock *create_block(int start, int size, int pid)
{
    MemoryBlock *block;

    block = (MemoryBlock *)malloc(sizeof(MemoryBlock));
    if (block == NULL)
    {
        return NULL;
    }

    block->start = start;
    block->size = size;
    block->pid = pid;
    block->prev = NULL;
    block->next = NULL;
    return block;
}

void insert_after(MemoryBlock *position, MemoryBlock *new_block)
{
    if (position == NULL || new_block == NULL)
    {
        return;
    }

    new_block->prev = position;
    new_block->next = position->next;
    position->next->prev = new_block;
    position->next = new_block;
}

void remove_block(MemoryBlock *removed)
{
    if (removed == NULL || removed->pid == HEAD_PID ||
        removed->prev == NULL || removed->next == NULL)
    {
        return;
    }

    removed->prev->next = removed->next;
    removed->next->prev = removed->prev;
    removed->prev = NULL;
    removed->next = NULL;
    free(removed);
}

void destroy_list(MemoryList *list)
{
    if (list == NULL)
    {
        return;
    }

    while (list->head.next != &list->head)
    {
        remove_block(list->head.next);
    }

    list->head.prev = &list->head;
    list->head.next = &list->head;
    list->total_size = 0;
}

void print_memory_map(const MemoryList *list)
{
    const MemoryBlock *block;

    if (list == NULL)
    {
        return;
    }

    printf("Memory map: ");
    block = list->head.next;
    while (block != &list->head)
    {
        printf("[%d-%d, ", block->start, block->start + block->size - 1);
        if (block->pid == FREE_PID)
        {
            printf("FREE]");
        }
        else
        {
            printf("P%d]", block->pid);
        }

        block = block->next;
        if (block != &list->head)
        {
            printf(" <-> ");
        }
    }
    printf("\n");
}

MemoryBlock *find_first_fit(MemoryList *list, int size)
{
    MemoryBlock *current;

    if (list == NULL || size <= 0)
    {
        return NULL;
    }

    current = list->head.next;
    while (current != &list->head)
    {
        if (current->pid == FREE_PID && current->size >= size)
        {
            return current;
        }
        current = current->next;
    }
    return NULL;
}

MemoryBlock *find_process_block(MemoryList *list, int pid)
{
    MemoryBlock *current;

    if (list == NULL || pid <= 0)
    {
        return NULL;
    }

    current = list->head.next;
    while (current != &list->head)
    {
        if (current->pid == pid)
        {
            return current;
        }
        current = current->next;
    }
    return NULL;
}

int allocate_memory(MemoryList *list, int pid, int size)
{
    if (list == NULL || pid <= 0 || size <= 0)
    {
        return 0;
    }
    if (find_process_block(list, pid) != NULL)
    {
        return 0;
    }

    MemoryBlock *block = find_first_fit(list, size);
    if (block == NULL)
    {
        return 0;
    }
    if (block->size == size)
    {
        block->pid = pid;
        return 1;
    }

    MemoryBlock *remaining = create_block(block->start + size,
                                          block->size - size, FREE_PID);
    if (remaining == NULL)
    {
        return 0;
    }
    block->size = size;
    block->pid = pid;
    insert_after(block, remaining);
    return 1;
}

MemoryBlock *merge_free_blocks(MemoryList *list, MemoryBlock *current)
{
    MemoryBlock *previous;
    MemoryBlock *next;

    if (list == NULL || current == NULL ||
        current == &list->head || current->pid != FREE_PID)
    {
        return current;
    }

    previous = current->prev;
    if (previous != &list->head && previous->pid == FREE_PID)
    {
        previous->size += current->size;
        remove_block(current);
        /* current가 삭제되었으므로 살아 있는 이전 노드를 기준으로 바꾼다. */
        current = previous;
    }

    next = current->next;
    if (next != &list->head && next->pid == FREE_PID)
    {
        current->size += next->size;
        remove_block(next);
    }

    return current;
}

int release_memory(MemoryList *list, int pid)
{
    if (list == NULL || pid <= 0)
    {
        return 0;
    }
    MemoryBlock *block = find_process_block(list, pid);
    if (block == NULL)
    {
        return 0;
    }
    block->pid = FREE_PID;
    merge_free_blocks(list, block);
    return 1;
}

int main(void)
{
    MemoryList memory;
    char command[COMMAND_LENGTH];
    int total_size;
    int command_count;
    int i;
    int pid;
    int size;
    int result;

    printf("전체 메모리 크기를 입력하세요: ");
    if (scanf("%d", &total_size) != 1 || total_size <= 0)
    {
        printf("잘못된 전체 메모리 크기입니다.\n");
        return 0;
    }

    if (!init_list(&memory, total_size))
    {
        printf("메모리 리스트를 초기화할 수 없습니다.\n");
        return 0;
    }

    printf("명령의 개수를 입력하세요: ");
    if (scanf("%d", &command_count) != 1 ||
        command_count < 0 || command_count > MAX_COMMANDS)
    {
        printf("잘못된 명령 개수입니다.\n");
        destroy_list(&memory);
        return 0;
    }

    for (i = 0; i < command_count; ++i)
    {
        if (scanf("%15s", command) != 1)
        {
            printf("명령을 읽을 수 없습니다.\n");
            break;
        }

        if (strcmp(command, "ALLOC") == 0)
        {
            if (scanf("%d %d", &pid, &size) != 2)
            {
                printf("ALLOC 명령 형식이 잘못되었습니다.\n");
                break;
            }
            result = allocate_memory(&memory, pid, size);
            printf("ALLOC P%d %d: %s\n",
                   pid, size, result ? "SUCCESS" : "FAIL");
        }
        else if (strcmp(command, "FREE") == 0)
        {
            if (scanf("%d", &pid) != 1)
            {
                printf("FREE 명령 형식이 잘못되었습니다.\n");
                break;
            }
            result = release_memory(&memory, pid);
            printf("FREE P%d: %s\n",
                   pid, result ? "SUCCESS" : "FAIL");
        }
        else
        {
            printf("알 수 없는 명령입니다: %s\n", command);
        }
        print_memory_map(&memory);
    }

    destroy_list(&memory);
    return 0;
}
