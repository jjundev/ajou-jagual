#define _CRT_SECURE_NO_WARNINGS

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define FREE_PID (-1)
#define HEAD_PID (-2)
#define MAX_COMMANDS 10000
#define COMMAND_LENGTH 16

기능 명세와 분석

제공된 헤더와 상수다. stdio.h는 scanf·printf, stdlib.h는 malloc·free, string.h는 strcmp에 사용된다. _CRT_SECURE_NO_WARNINGS는 MSVC의 관련 보안 경고를 억제하는 매크로이며 입력을 검증하거나 안전성을 보장하는 기능은 아니다.

FREE_PID(-1)는 빈 블록, HEAD_PID(-2)는 헤드의 표식이다. 실제 PID는 양수다. MAX_COMMANDS(10000)는 명령 개수 상한이고, COMMAND_LENGTH(16)는 명령 문자열의 저장 공간이다. scanf의 %15s와 결합해 널 문자 한 칸을 남긴다.

typedef struct MemoryBlock
{
    int start;
    int size;
    int pid;
    struct MemoryBlock *prev;
    struct MemoryBlock *next;
} MemoryBlock;

기능 명세와 분석

제공된 MemoryBlock은 한 연속 구간의 시작 주소(start), 칸 수(size), 소유자(pid), 앞뒤 링크(prev·next)를 보관한다. 끝 주소는 start + size - 1이다.

노드의 앞뒤 순서는 주소 순서와 일치한다. 빈 구간도 FREE_PID를 가진 노드로 기록하므로 사용 중인 구간과 빈 구간을 모두 리스트에서 추적할 수 있다.

typedef struct
{
    MemoryBlock head;
    int total_size;

기능 명세와 분석

제공된 MemoryList는 기준 노드 head 자체와 전체 크기 total_size를 포함한다. head는 실제 할당 대상이 아니고 원형 리스트의 시작과 끝을 구분한다.

실제 노드가 하나도 없을 때 head.prev와 head.next는 head 자신을 가리킨다. 별도의 파일 범위 전역 상태 변수는 없으며 main의 지역 변수 memory를 포인터로 전달해 상태를 공유한다.

init_list

제공 함수

소스코드기능 명세와 분석
 
int init_list(MemoryList *list, int total_size)
{
    MemoryBlock *initial_block;

    if (list == NULL || total_size <= 0)
    {
        return 0;
    }

    list->head.start = -1;
    list->head.size = 0;
    list->head.pid = HEAD_PID;
    list->head.prev = &list->head;
    list->head.next = &list->head;
    list->total_size = total_size;

    initial_block = create_block(0, total_size, FREE_PID);
    if (initial_block == NULL)
    {
        list->total_size = 0;
        return 0;
    }

    insert_after(&list->head, initial_block);
    return 1;
}

기능 명세

인자: 초기화할 MemoryList 포인터와 전체 크기. 반환값: 성공 1, 잘못된 인자나 초기 노드 생성 실패 시 0. 변경 상태: 헤드의 연결, total_size, 초기 FREE 블록.

분석

헤드는 실제 주소 구간을 나타내지 않는 기준 노드다. 먼저 prev와 next를 자기 자신으로 설정해 빈 원형 리스트를 만든다. 이후 시작 주소 0, 크기 total_size인 FREE 블록 하나를 생성해 헤드 뒤에 연결한다.

초기 블록을 만들지 못하면 total_size를 0으로 되돌리고 실패를 반환한다. 헤드만 자기 자신을 가리키는 상태이므로 불완전한 실제 블록이 리스트에 남지 않는다.

create_block

제공 함수

소스코드기능 명세와 분석
 
MemoryBlock *create_block(int start, int size, int pid)
{
    MemoryBlock *block;

    block = (MemoryBlock *)malloc(sizeof(MemoryBlock));
    if (block == NULL)
    {
        return NULL;
    }

    block->start = start;
    block->size = size;
    block->pid = pid;
    block->prev = NULL;
    block->next = NULL;
    return block;
}

기능 명세

인자: 시작 주소, 크기, PID. 반환값: 생성한 MemoryBlock 포인터 또는 NULL. 변경 상태: 성공 시 노드 한 개를 동적 생성한다. 리스트 연결은 변경하지 않는다.

분석

malloc으로 확보하는 것은 프로세스가 사용할 size칸의 실제 RAM이 아니라, start·size·pid와 링크를 보관하는 관리 노드의 저장 공간이다. 따라서 요청량과 관계없이 sizeof(MemoryBlock)만큼 생성한다.

prev와 next는 NULL로 초기화한다. 아직 어느 리스트에도 삽입되지 않았다는 상태이며, 실제 연결은 insert_after가 담당한다. malloc 실패 시 멤버에 접근하지 않고 NULL을 반환한다.

insert_after

제공 함수

소스코드기능 명세와 분석
 
void insert_after(MemoryBlock *position, MemoryBlock *new_block)
{
    if (position == NULL || new_block == NULL)
    {
        return;
    }

    new_block->prev = position;
    new_block->next = position->next;
    position->next->prev = new_block;
    position->next = new_block;
}

기능 명세

인자: 삽입 기준 노드 position과 새 노드 new_block. 반환값: 없음. 변경 상태: 두 노드와 기존 다음 노드의 prev·next 연결.

분석

position과 원래 다음 노드 사이에 새 노드를 넣는다. 새 노드의 prev는 position, next는 원래 다음 노드를 가리키게 한다. 이어서 원래 다음 노드의 prev와 position의 next를 새 노드로 바꾼다.

position->next를 마지막에 바꾸므로 원래 다음 노드를 잃지 않고 네 방향 연결을 완성할 수 있다. 꼬리 뒤에 삽입할 때는 원래 다음 노드가 헤드이므로 헤드의 prev도 함께 갱신된다. 유효한 연결 리스트와 아직 연결되지 않은 새 노드를 인자로 받는다는 전제다.

remove_block

제공 함수

소스코드기능 명세와 분석
 
void remove_block(MemoryBlock *removed)
{
    if (removed == NULL || removed->pid == HEAD_PID ||
        removed->prev == NULL || removed->next == NULL)
    {
        return;
    }

    removed->prev->next = removed->next;
    removed->next->prev = removed->prev;
    removed->prev = NULL;
    removed->next = NULL;
    free(removed);
}

기능 명세

인자: 제거할 노드 removed. 반환값: 없음. 변경 상태: 양옆 노드를 직접 연결하고 removed의 동적 저장 공간을 해제한다.

분석

NULL, 헤드 PID, 연결되지 않은 노드는 제거하지 않는다. 정상 노드는 이전 노드의 next와 다음 노드의 prev가 서로를 가리키게 바꾼 뒤 free한다. 원형 구조에서 첫 노드나 마지막 노드를 지워도 이웃인 헤드의 링크가 같은 방식으로 바뀐다.

이 함수 호출 후 기존 removed 포인터로 노드 멤버를 읽으면 안 된다. 또한 이 함수 자체는 가상 주소 구간을 이웃 블록에 합치지 않는다. 병합 함수가 먼저 크기를 합산한 뒤 불필요해진 관리 노드만 삭제해야 전체 구간이 보존된다.

void destroy_list(MemoryList *list)
{
    if (list == NULL)
    {
        return;
    }

    while (list->head.next != &list->head)
    {
        remove_block(list->head.next);
    }

    list->head.prev = &list->head;
    list->head.next = &list->head;
    list->total_size = 0;
}

기능 명세

인자: 정리할 리스트. 반환값: 없음. 변경 상태: 모든 실제 노드를 해제하고 헤드 연결과 total_size를 초기 상태로 정리한다.

분석

헤드 다음 노드가 헤드 자체가 될 때까지 첫 실제 노드를 반복해서 제거한다. 삭제된 노드의 next를 따라가지 않고 매번 list->head.next를 다시 읽기 때문에 해제된 노드를 다음 반복에서 참조하지 않는다.

헤드는 MemoryList 내부에 포함된 객체이므로 free하지 않는다. 실제 노드를 모두 없앤 뒤 prev·next를 헤드 자신으로 맞추고 total_size를 0으로 설정한다. 노드 수를 n이라 하면 전체 정리에는 O(n)이 필요하다.

print_memory_map

제공 함수

소스코드기능 명세와 분석
 
void print_memory_map(const MemoryList *list)
{
    const MemoryBlock *block;

    if (list == NULL)
    {
        return;
    }

    printf("Memory map: ");
    block = list->head.next;
    while (block != &list->head)
    {
        printf("[%d-%d, ", block->start, block->start + block->size - 1);
        if (block->pid == FREE_PID)
        {
            printf("FREE]");
        }
        else
        {
            printf("P%d]", block->pid);
        }

        block = block->next;
        if (block != &list->head)
        {
            printf(" <-> ");
        }
    }
    printf("\n");
}

기능 명세

인자: 읽기 전용 리스트 포인터. 반환값: 없음. 변경 상태: 리스트를 변경하지 않고 표준 출력에 메모리 지도를 쓴다.

분석

헤드 다음부터 실제 블록을 순회한다. 각 블록의 마지막 주소는 start + size - 1이다. 예를 들어 start=0, size=8이면 주소는 0~7이며, 시작과 끝을 모두 포함하므로 8칸이다.

FREE_PID이면 FREE, 아니면 P와 PID를 출력한다. 다음 노드가 헤드가 아닐 때만 연결 표시를 넣어 마지막에 불필요한 구분자가 남지 않게 한다. 탐색 중 노드가 헤드로 돌아오면 한 바퀴를 마친 것이다. 노드 수에 비례해 O(n)의 순회가 필요하다.

find_first_fit

구현 대상 함수

소스코드기능 명세와 분석
 
MemoryBlock *find_first_fit(MemoryList *list, int size)
{
    MemoryBlock *current;

    if (list == NULL || size <= 0)
    {
        return NULL;
    }

    current = list->head.next;
    while (current != &list->head)
    {
        if (current->pid == FREE_PID && current->size >= size)
        {
            return current;
        }
        current = current->next;
    }
    return NULL;
}

기능 명세

인자: 리스트와 요청 크기. 반환값: 조건을 만족하는 첫 FREE 블록의 포인터, 없거나 인자가 잘못되면 NULL. 변경 상태: 없음.

분석

앞에서부터 pid == FREE_PID와 size >= 요청 크기를 동시에 만족하는 노드를 찾는다. 만족하면 즉시 반환하므로 뒤에 더 작은 적합 블록이 있어도 비교하지 않는다. 빈 공간의 합이 충분한지는 판단 조건이 아니다.

원형 리스트에서는 마지막 노드의 next가 NULL이 아니라 헤드다. 따라서 current != &list->head로 종료하며 헤드를 실제 후보로 검사하지 않는다. 최악에는 모든 실제 블록을 확인하므로 O(n)이다. 크기 15와 10의 빈 블록이 이 순서로 있고 8칸을 요청하면 앞의 15칸을 선택한다.

MemoryBlock *find_process_block(MemoryList *list, int pid)
{
    MemoryBlock *current;

    if (list == NULL || pid <= 0)
    {
        return NULL;
    }

    current = list->head.next;
    while (current != &list->head)
    {
        if (current->pid == pid)
        {
            return current;
        }
        current = current->next;
    }
    return NULL;
}

기능 명세

인자: 리스트와 양의 PID. 반환값: 해당 PID를 가진 블록 포인터, 없거나 인자가 잘못되면 NULL. 변경 상태: 없음.

분석

헤드 다음부터 한 바퀴 동안 pid가 입력과 같은 노드를 탐색한다. PID가 양수인 경우만 찾으므로 FREE_PID(-1)와 HEAD_PID(-2)를 프로세스로 반환하지 않는다.

할당에서는 같은 PID가 이미 존재하는지 확인하는 데, 반납에서는 반납할 블록을 찾는 데 재사용한다. 현재 구현은 PID 한 개에 블록 한 개를 대응시키며 중복 할당을 거부한다. 최악의 탐색 시간은 O(n)이다.

    if (block->size > size)
    {
        /* 새 노드 생성에 실패해도 기존 리스트는 바뀌지 않도록 먼저 생성한다. */
        remaining = create_block(block->start + size,
                                 block->size - size, FREE_PID);
        if (remaining == NULL)
        {
            return 0;
        }

        block->size = size;
        insert_after(block, remaining);
    }

    block->pid = pid;
    return 1;
}

분석 (계속)

선택한 블록이 더 크면 남는 블록의 시작 주소는 기존 start + 요청 크기, 남는 크기는 기존 size - 요청 크기다. 먼저 이 정보를 가진 새 노드를 만든다. 생성 실패 시 기존 노드를 바꾸지 않아 원래 상태가 유지된다.

새 노드 생성에 성공한 뒤 기존 크기를 요청량으로 줄이고 남은 FREE 노드를 바로 뒤에 삽입한다. 크기가 정확히 같으면 분할과 새 노드 생성 없이 기존 노드의 PID만 바꾼다. 마지막에 성공 1을 반환한다. PID 탐색과 First-Fit 탐색을 하므로 최악 시간은 O(n)이며, 이를 실제 속도 향상 측정으로 해석하지 않는다.

merge_free_blocks

구현 대상 함수

소스코드기능 명세와 분석
 
MemoryBlock *merge_free_blocks(MemoryList *list, MemoryBlock *current)
{
    MemoryBlock *previous;
    MemoryBlock *next;

    if (list == NULL || current == NULL ||
        current == &list->head || current->pid != FREE_PID)
    {
        return current;
    }

    previous = current->prev;
    if (previous != &list->head && previous->pid == FREE_PID)
    {
        previous->size += current->size;
        remove_block(current);
        /* current가 삭제되었으므로 살아 있는 이전 노드를 기준으로 바꾼다. */
        current = previous;
    }

기능 명세

인자: 리스트와 병합 기준 FREE 노드. 반환값: 병합 후 살아 있는 기준 노드. 변경 상태: 인접 FREE 구간의 크기와 링크, 중복 관리 노드의 삭제.

분석

잘못된 인자, 헤드, 사용 중 블록은 병합하지 않는다. 이전 노드가 실제 FREE 블록이면 previous의 크기에 current의 크기를 더하고 current 노드를 삭제한다. 이어 current = previous로 바꾸어 해제되지 않은 노드를 기준으로 삼는다. 이 대입이 없으면 다음 병합에서 삭제된 노드를 참조하게 된다.

    next = current->next;
    if (next != &list->head && next->pid == FREE_PID)
    {
        current->size += next->size;
        remove_block(next);
    }

    return current;
}

분석 (계속)

이전 병합이 끝난 기준 노드의 next를 다시 읽어 오른쪽 FREE 블록도 합친다. 오른쪽 노드의 크기를 더한 뒤 그 노드를 삭제하므로 시작 주소는 가장 왼쪽 주소로 유지된다. 헤드는 제외하므로 리스트가 원형이라는 이유로 메모리의 끝과 처음을 붙이지 않는다.

정상적인 할당·반납 뒤에는 인접 FREE 블록이 남지 않는 상태가 유지된다. 이 전제에서 한 블록을 반납할 때 양옆만 검사하면 되므로 이 함수 자체는 O(1)이다. 예를 들어 [0~19 FREE], [20~49 FREE], [50~99 FREE]는 시작 0·크기 100인 한 블록으로 합쳐진다.

release_memory

구현 대상 함수

소스코드기능 명세와 분석
 
int release_memory(MemoryList *list, int pid)
{
    MemoryBlock *block;

    block = find_process_block(list, pid);
    if (block == NULL)
    {
        return 0;
    }

    block->pid = FREE_PID;
    merge_free_blocks(list, block);
    return 1;
}

기능 명세

인자: 리스트와 반납할 PID. 반환값: 해당 블록 반납 성공 시 1, 찾지 못하면 0. 변경 상태: 블록을 FREE로 표시하고 가능한 이웃 FREE 블록과 병합한다.

분석

PID를 찾지 못하면 리스트를 변경하지 않고 실패한다. 찾으면 pid를 FREE_PID로 바꾼다. 이는 프로세스에 배정한 가상 구간을 사용 가능한 공간으로 돌려놓는 처리다.

FREE 처리만으로 관리 노드를 즉시 삭제하면 해당 주소 구간의 기록이 사라진다. 따라서 FREE 노드는 유지하고 merge_free_blocks로 이웃을 합칠 때만 불필요한 노드를 제거한다. 양옆이 사용 중이면 현재 노드가 그대로 남는다. PID 탐색 O(n)과 병합 O(1)을 합쳐 최악 시간은 O(n)이다.

MemoryBlock *merge_free_blocks(MemoryList *list, MemoryBlock *current)
{
    MemoryBlock *previous;
    MemoryBlock *next;
