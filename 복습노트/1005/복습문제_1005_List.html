#include <stdlib.h>

typedef struct Node { int data; struct Node *link; } Node;
int count_nodes(Node *head) {

    for (Node *p = head; p != NULL; p = p->link) ++count;
    return count;
}
int main(void) {
    int n, value;
    Node *head = NULL, *tail = NULL;
    if (scanf("%d", &n) != 1 || n < 0) return 1;
    for (int i = 0; i < n; ++i) {
        if (scanf("%d", &value) != 1) return 1;
        Node *p = malloc(sizeof *p);
        if (p == NULL) return 1;
        p->data = value;
        p->link = NULL;
        if (tail == NULL) head = p;
        else tail->link = p;
        tail = p;
    }
    printf("%d\n", count_nodes(head));
    while (head != NULL) {
        Node *next = head->link;
        free(head);
        head = next;
    }
    return 0;
}

P3. 두 연결 리스트를 이어 붙일 때

강의 예제 기반 자체 출제 S4 p31

서로 노드를 공유하지 않는 두 단순 연결 리스트 head1 = A -> B -> NULL, head2 = C -> D -> NULL에 대해 아래 concat 함수를 호출해요.

ListNode *concat(ListNode *head1,ListNode *head2)
{
    ListNode *p;
    if(head1==NULL) return head2;
    else if(head2==NULL) return head1;
    else {
        p=head1;
        while(p->link!=NULL) p=p->link;
        p->link=head2;
        return head1;
    }
}
  1. (a) concat(head1, head2) 실행 시 포인터 p가 이동을 멈추는 노드, 새로 연결되는 링크 대입식(p->link = head2), 반환 포인터 및 결과 체인을 쓰세요.
  2. (b) head1 == NULL일 때의 반환값과, 반복 조건을 while (p != NULL)로 잘못 썼을 때 생기는 오류 및 첫 리스트 길이 $m$에 대한 시간 복잡도를 설명하세요.

풀이 보기
  1. (a) p = A에서 시작해 p->link != NULL인 동안 이동하면 마지막 노드 p = B(B->link == NULL)에서 멈춰요. 그다음 B->link = C(p->link = head2)로 연결하고 head1 = A를 반환하므로 결과 체인은 A -> B -> C -> D -> NULL이에요.
  2. (b) 첫 리스트가 비어 있으면(head1 == NULL) 첫 분기에서 곧바로 head2를 반환해요. 만약 while (p != NULL)로 쓰면 루프가 끝났을 때 p == NULL이 되어 p->link = head2에서 널 포인터 역참조 오류가 발생하며, 첫 리스트의 마지막 노드를 찾기 위해 길이 $m$만큼 순회하므로 시간 복잡도는 $O(m)$이에요.

정답: (a) p는 마지막 노드 B에서 멈추고 B->link = C로 변경, 반환 포인터 head1 = A, 결과 체인 A -> B -> C -> D -> NULL (b) head1 == NULL이면 head2를 반환하며, while (p != NULL)로 쓰면 루프 종료 시 p == NULL에서 p->link를 역참조하는 오류가 난다. 시간 복잡도는 $O(m)$이다.

단순 이어 붙이기는 정렬 병합이 아니며 두 리스트를 값 순으로 섞지 않아요.

P4. 역순 코드의 세 포인터

강의 예제 기반 자체 출제 S4 p32

단순 연결 리스트 head = A -> B -> C -> NULL에 대해 아래 reverse(head) 함수를 호출해요.

ListNode *reverse(ListNode *head)
{
    ListNode *p,*q,*r;
    p=head; q=NULL;
    while(p!=NULL) {
        r=q;
        q=p;
        p=p->link;
        q->link=r;
    }
    return q;
}
  1. (a) while 루프 1~3회차가 끝난 직후마다 포인터 (r, q, p)가 가리키는 노드와 이미 역순으로 뒤집힌 부분 체인(q부터 시작하는 체인)을 쓰세요.
  2. (b) 루프 안에서 q = p; p = p->link; 전에 q->link = r;를 먼저 실행하면 안 되는 이유와, 공백 리스트(head == NULL) 입력 시 반환값을 설명하세요.

p는 아직 처리하지 않은 앞부분, q는 방금 뒤집을 현재 노드, r은 이미 뒤집힌 부분의 첫 노드예요.

풀이 보기
  1. (a) 1회차 직후 (r, q, p) = (NULL, A, B)이고 뒤집힌 체인은 A -> NULL이에요. 2회차 직후 (r, q, p) = (A, B, C)이고 뒤집힌 체인은 B -> A -> NULL이에요. 3회차 직후 (r, q, p) = (B, C, NULL)이고 뒤집힌 체인은 C -> B -> A -> NULL이 되어 루프가 끝나고 q = C를 반환해요.
  2. (b) p = p->link로 다음 노드를 대피시키기 전에 q->link = r을 먼저 실행하면(이때 q == p) 원래 다음 노드로 가는 링크가 r로 덮어써져 나머지 리스트를 잃어버려요. 공백 리스트(head == NULL)가 입력되면 while (p != NULL)이 한 번도 실행되지 않고 초기값 q = NULL을 그대로 반환해요.

정답: (a) 1회차: (r, q, p) = (NULL, A, B), A -> NULL; 2회차: (A, B, C), B -> A -> NULL; 3회차: (B, C, NULL), C -> B -> A -> NULL (반환값 C) (b) p = p->link 전에 q->link = r을 실행하면 q와 p가 같은 노드를 가리키고 있어 원래 다음 노드 주소를 잃어버린다. 공백 리스트 입력 시 반환값은 NULL이다.

원래 head 변수가 자동으로 새 첫 노드를 가리키는 것은 아니므로 호출자가 반환값을 받아야 해요.

P5. 연결 리스트 삽입의 세 경우

강의 예제 기반 자체 출제 S4 p25

새로 동적 할당한 노드 new_node = N을 단순 연결 리스트에 삽입하는 아래 insert_node 함수를 세 가지 독립적인 상황에서 호출해요.

void insert_node(ListNode **phead,ListNode *p,ListNode *new_node)
{
    if(*phead==NULL) {
        new_node->link=NULL;
        *phead=new_node;
    } else if(p==NULL) {
        new_node->link=*phead;
        *phead=new_node;
    } else {
        new_node->link=p->link;
        p->link=new_node;
    }
}
  1. (a) 공백 리스트(*phead == NULL), A -> B -> NULL에서 맨 앞 삽입(p == NULL), 같은 리스트에서 중간 삽입(p == A) 세 경우 각각에 대해 실행되는 링크 대입과 결과 체인을 쓰세요.
  2. (b) 마지막 분기(else)에서 두 대입문 new_node->link = p->link;와 p->link = new_node;의 순서를 바꾸면 뒷부분 노드를 잃어버리는 이유와, 매개변수로 이중 포인터 ListNode **phead를 쓰는 이유를 설명하세요.
강의 예제 보기

힌트 보기

*phead는 호출자의 head 변수 자체이며, 중간 삽입에서는 p->link가 가리키던 기존 뒷노드를 new_node->link에 먼저 연결해야 해요.

풀이 보기
  1. (a) ① 공백 리스트(*phead == NULL): N->link = NULL; *phead = N;이 실행되어 N -> NULL이 돼요. ② 맨 앞 삽입(p == NULL): N->link = A; *phead = N;이 실행되어 N -> A -> B -> NULL이 돼요. ③ 중간 삽입(p == A): N->link = B; A->link = N;이 실행되어 A -> N -> B -> NULL이 돼요.
  2. (b) p->link = new_node;를 먼저 실행하면 p->link가 N으로 덮어써진 뒤 new_node->link = p->link;가 실행되어 N->link = N(자기 순환)이 되고 기존 뒷노드 B로 가는 주소를 잃어버려요. 또한 공백 삽입이나 맨 앞 삽입에서는 호출자의 head 포인터 변수 값 자체를 새 노드 주소로 갱신해야 하므로 이중 포인터 ListNode **phead를 사용해요.

정답: (a) 공백 리스트: N -> NULL; p == NULL(맨 앞 삽입): N -> A -> B -> NULL; p == A(중간 삽입): A -> N -> B -> NULL (b) 대입 순서를 바꾸면 p->link가 먼저 N으로 덮어써져 N->link = N이 되고 기존 뒷노드 B를 잃어버린다. 공백·맨 앞 삽입 시 호출자의 head 포인터 변수 값을 직접 변경하기 위해 이중 포인터 ListNode **phead를 쓴다.

p == NULL은 리스트가 비어 있다는 뜻이 아니라 선행 노드가 없어 맨 앞에 삽입한다는 뜻이에요.

P6. 삭제할 노드를 해제하기 전의 연결

강의 예제 기반 자체 출제 S4 p28

각 노드를 malloc으로 할당한 단순 연결 리스트 head = A -> B -> C -> NULL에서 아래 remove_node 함수를 호출해요.

void remove_node(ListNode **phead,ListNode *p,ListNode *removed)
{
    if(p==NULL) *phead=(*phead)->link;
    else p->link=removed->link;
    free(removed);
}
  1. (a) 중간 노드 B를 삭제할 때와 맨 앞 노드 A를 삭제할 때 각각 전달할 (p, removed) 인자와 삭제 후 체인을 쓰고, 노드가 A 하나뿐인 리스트에서 A를 삭제한 뒤 head 값을 쓰세요.
  2. (b) p == NULL(맨 앞 삭제) 분기가 따로 필요한 이유와, 링크 갱신 전에 free(removed)를 먼저 호출하면 안 되는 이유를 설명하세요.

p는 삭제 대상의 직전(선행) 노드이고 removed는 실제로 삭제·반납할 노드예요.

풀이 보기
  1. (a) B 삭제 시에는 선행 노드가 A이므로 (p = A, removed = B)를 전달해 A->link = C가 되고 결과는 A -> C -> NULL이에요. 맨 앞 A 삭제 시에는 선행 노드가 없으므로 (p = NULL, removed = A)를 전달해 head = B가 되고 결과는 B -> C -> NULL이에요. 단일 노드 A -> NULL에서 A를 삭제하면 *phead = A->link = NULL이 되어 head = NULL이 돼요.
  2. (b) 맨 앞 노드는 선행 노드 p가 존재하지 않아 p->link가 아니라 시작 포인터 *phead 자체를 갱신해야 하므로 p == NULL 분기가 따로 필요해요. 또한 free(removed)를 먼저 호출하면 이미 반납된 메모리의 removed->link를 읽게 되는(use-after-free) 오류가 발생하므로 반드시 링크를 먼저 이어 준 뒤 free(removed)를 호출해야 해요.

정답: (a) B 삭제: p = A, removed = B, 결과 A -> C -> NULL; A 삭제: p = NULL, removed = A, 결과 B -> C -> NULL; 단일 노드 삭제 후 head = NULL (b) 맨 앞 노드는 선행 노드 p가 없어 *phead 자체를 갱신해야 하므로 p == NULL 분기가 필요하고, free(removed)를 먼저 실행하면 해제된 메모리의 removed->link를 읽는 오류가 발생한다.

단순 연결 리스트에서는 삭제 대상 포인터 removed만으로는 직전 노드 p의 link를 고칠 수 없어 선행 노드 포인터 p가 함께 필요해요.

P7. 꼬리를 head로 두는 원형 리스트

강의 예제 기반 자체 출제 S4 p35

head가 마지막(꼬리) 노드를 가리키고 head->link가 첫 노드를 가리키는 원형 연결 리스트(현재 A -> B -> A, head = B)에 새 노드 N을 삽입해요.

void insert_last(ListNode **phead,ListNode *node)
{
    if(*phead==NULL) {
        *phead=node; node->link=node;
    } else {
        node->link=(*phead)->link;
        (*phead)->link=node;
        *phead=node;
    }
}
  1. (a) insert_last(&head, N) 실행 후 각 노드의 link(A->link, B->link, N->link)와 head 및 첫 노드를 쓰세요.
  2. (b) 마지막 줄 *phead = node;를 생략하면 논리적으로 어떤 위치 삽입이 되는지 쓰고, head가 꼬리 노드를 가리키게 설계할 때 맨 앞(insert_first)·맨 뒤(insert_last) 삽입이 모두 $O(1)$이 되는 이유를 설명하세요.

P8. 이중 연결 리스트의 삽입과 삭제

P10. 헤드 포인터와 헤드 노드의 공백 표현

첫 비교 p = (3, 12), q = (8, 12)는 지수가 같으므로 계수를 더한 (11, 12)를 붙이고 p, q를 모두 다음 노드로 옮겨요.

풀이 보기
  1. (a) ① 지수 12 == 12: (11, 12)를 추가하고 p, q 모두 이동해요. ② 지수 8 < 10: (-3, 10)을 추가하고 q만 이동해요. ③ 지수 8 > 6: (2, 8)을 추가하고 p만 이동해요. ④ 지수 0 < 6: (10, 6)을 추가하고 q가 NULL이 되어 B가 끝나요. ⑤ 남은 A의 (1, 0)을 그대로 복사해 결과는 $11x^{12} - 3x^{10} + 2x^8 + 10x^6 + 1$이에요.
  2. (b) 같은 지수의 계수 합이 0이면 결과 다항식에 항을 추가하지 않지만 두 입력 항은 모두 처리했으므로 p와 q를 둘 다 다음 노드로 이동해야 해요. 매 단계마다 적어도 한 포인터가 전진하여 각 항을 정확히 한 번씩만 소비하므로 전체 시간 복잡도는 $O(m + n)$이에요.

정답: (a) 12 == 12에서 (11, 12) 추가 후 p, q 모두 이동 → 8 < 10에서 (-3, 10) 추가 후 q 이동 → 8 > 6에서 (2, 8) 추가 후 p 이동 → 0 < 6에서 (10, 6) 추가 후 q 이동(종료) → 남은 A의 (1, 0) 복사, 최종 다항식 $11x^{12} - 3x^{10} + 2x^8 + 10x^6 + 1$ (b) 같은 지수의 계수 합이 0이면 결과에 노드를 추가하지 않되 p, q는 모두 이동한다. 두 리스트의 각 항을 한 번씩만 순회하므로 시간 복잡도는 $O(m + n)$이다.

계수가 큰 쪽이 아니라 지수가 큰 쪽을 먼저 선택하며, 한쪽 리스트가 끝났다고 다른 쪽에 남은 항을 버리지 않아요.
