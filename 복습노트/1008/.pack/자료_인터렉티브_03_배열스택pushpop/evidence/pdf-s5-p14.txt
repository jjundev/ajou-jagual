        void push(StackType *s, element item)
        {

                   if( is_full(s) ) {
                               fprintf(stderr,"  \n");
                                return;

                     }
                     else s->stack[++(s->top)] = item;
        }
        // 
        element pop(StackType *s)
        {
               if( is_empty(s) ) {

                               fprintf(stderr, "  \n");
                                exit(1);
                 }
                 else return s->stack[(s->top)--];
        }
        // 
        element peek(StackType *s)
        {
               if( is_empty(s) ) {
                               fprintf(stderr, "  \n");
                                exit(1);
                 }
                 else return s->stack[s->top];
        }
