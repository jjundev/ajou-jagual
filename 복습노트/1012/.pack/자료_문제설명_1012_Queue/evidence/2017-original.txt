10. 다음의 push, pop, main 함수 코드를 완성하시오. [15 points]
#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#define MAX_STACK_SIZE 100


typedef int element;
typedef struct {
          element stack[MAX_STACK_SIZE];
          int top;
} StackType;


void init(StackType *s){      // 스택초기화함수
          s->top = -1;
}


int is_empty(StackType *s){ // 공백상태검출함수
          return (s->top == -1);
}


int is_full(StackType *s){    // 포화상태검출함수
          return (s->top == (MAX_STACK_SIZE - 1));
}


void push(StackType *s, element item){
          if (is_full(s)) {
                       fprintf(stderr, "스택 포화 에러\n");
          } else {
                       s->top = s->top + 1;
                       s->stack[s->top] = item;
          }
}
element pop(StackType *s) {
          element item;


          if (is_empty(s)) {
                       fprintf(stderr, "스택 공백 에러\n");
          }
          else {
                       item = s->stack[s->top];
                       s->top = s->top - 1;
                       return item;
          }
}
구체적으로 main 함수에서는 스택을 이용하여 후위 표기식을 계산하는 작업을 하며 후위 표
기식은 한 자리 숫자(0~9)들과 연산자 (+, *)만을 사용하며 공백이 없이 입력을 한다. 후위 표
기식 계산을 위한 for문 안에 필요한 코드를 작성하시오.
int main(){
        char str[100];
        int i, len, op1, op2;
        int op;
        StackType s;


        init(&s);
        printf("input postfix equation using 0~9,+,*:");
        scanf("%s", str);
        len = strlen(str);



        for (i = 0; i < len; i++){
                    if (str[i] == '+'){
                             op1 = pop(&s);
                             op2 = pop(&s);
                             push(&s, op1 + op2);
                    }
                    else if (str[i] == '*'){
                             op1 = pop(&s);
                             op2 = pop(&s);
                             push(&s, op1 * op2);
                    }
                    else {
                             op = str[i] - '0';
                             push(&s, op);
                    }
        } // end of for
        printf("result:%d", pop(&s));
        return 0;
}
