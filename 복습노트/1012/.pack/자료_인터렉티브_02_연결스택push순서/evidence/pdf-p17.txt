void push(LinkedStackType *s, element item)
{

     StackNode *temp=(StackNode *)malloc(sizeof(StackNode));
     if( temp == NULL ){

          fprintf(stderr, " \n");
          return;
     }
     else{
          temp->item = item;
          temp->link = s->top;
          s->top = temp;
     }
}
